// ===================================================== // CONTRACTOR ENQUIRIES PORTAL // GOOGLE SHEETS DATABASE + APPS SCRIPT API // ===================================================== const SPREADSHEET_ID = "1bbw3SfSyP5iZoPylhhJTgMF_rzH50GfPoyvpxrKMHA4"; // ===================================================== // SHEET STRUCTURE // ===================================================== const HEADERS = { Enquiries: [ "id", "name", "phone", "workType", "location", "budget", "requiredDate", "description", "imageUrl", "date", "status" ], Responses: [ "id", "enquiryId", "contractor", "phone", "message", "date", "status" ], Contractors: [ "id", "contractorName", "company", "location", "workType", "experience", "phone", "whatsapp", "photoUrl", "status", "date" ] }; // ===================================================== // SETUP SHEETS // RUN THIS FUNCTION ONCE // ===================================================== function setupSheets() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); Object.keys(HEADERS).forEach(function(sheetName) { let sheet = ss.getSheetByName(sheetName); // Create sheet if it doesn't exist if (!sheet) { sheet = ss.insertSheet(sheetName); } // Add headers if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS[sheetName]); } }); return jsonResponse({ success: true, message: "Contractor portal sheets are ready." }); } // ===================================================== // GET API // ===================================================== function doGet(e) { try { const action = e && e.parameter ? (e.parameter.action || "getEnquiries").toLowerCase() : "getenquiries"; // --------------------------------------------- // GET ENQUIRIES // --------------------------------------------- if (action === "getenquiries") { const enquiries = getSheetData("Enquiries"); return jsonResponse({ success: true, enquiries: enquiries }); } // --------------------------------------------- // GET CONTRACTORS // --------------------------------------------- if (action === "getcontractors") { const contractors = getSheetData("Contractors"); return jsonResponse({ success: true, contractors: contractors }); } // --------------------------------------------- // GET RESPONSES // --------------------------------------------- if (action === "getresponses") { const responses = getSheetData("Responses"); return jsonResponse({ success: true, responses: responses }); } // --------------------------------------------- // GET EVERYTHING // --------------------------------------------- if (action === "getall") { return jsonResponse({ success: true, enquiries: getSheetData("Enquiries"), contractors: getSheetData("Contractors"), responses: getSheetData("Responses") }); } return jsonResponse({ success: false, error: "Invalid action." }); } catch (error) { return jsonResponse({ success: false, error: error.toString() }); } } // ===================================================== // POST API // ===================================================== function doPost(e) { try { if (!e || !e.postData) { return jsonResponse({ success: false, error: "No POST data received." }); } const data = JSON.parse(e.postData.contents); // ================================================= // NEW ENQUIRY // ================================================= if (data.action === "postEnquiry") { const enquiry = { id: data.id || ("ENQ" + new Date().getTime()), name: data.name || "", phone: data.phone || "", workType: data.workType || "", location: data.location || "", budget: data.budget || "", requiredDate: data.requiredDate || "", description: data.description || "", imageUrl: data.imageUrl || "", date: data.date || getToday(), // New enquiries need admin approval status: "Pending" }; appendToSheet( "Enquiries", enquiry ); return jsonResponse({ success: true, message: "Enquiry submitted successfully.", id: enquiry.id }); } // ================================================= // CONTRACTOR RESPONSE // ================================================= if (data.action === "sendResponse") { const response = { id: data.id || ("RES" + new Date().getTime()), enquiryId: data.enquiryId || "", contractor: data.contractor || "", phone: data.phone || "", message: data.message || "", date: data.date || getToday(), status: "New" }; appendToSheet( "Responses", response ); return jsonResponse({ success: true, message: "Response submitted successfully.", id: response.id }); } // ================================================= // NEW CONTRACTOR // ================================================= if (data.action === "addContractor") { const contractor = { id: data.id || ("CON" + new Date().getTime()), contractorName: data.contractorName || "", company: data.company || "", location: data.location || "", workType: data.workType || "", experience: data.experience || "", phone: data.phone || "", whatsapp: data.whatsapp || "", photoUrl: data.photoUrl || "", status: "Pending", date: data.date || getToday() }; appendToSheet( "Contractors", contractor ); return jsonResponse({ success: true, message: "Contractor profile submitted.", id: contractor.id }); } return jsonResponse({ success: false, error: "Unknown action." }); } catch (error) { return jsonResponse({ success: false, error: error.toString() }); } } // ===================================================== // READ GOOGLE SHEET // ===================================================== function getSheetData(sheetName) { const ss = SpreadsheetApp.openById( SPREADSHEET_ID ); const sheet = ss.getSheetByName( sheetName ); if (!sheet) { return []; } const lastRow = sheet.getLastRow(); if (lastRow < 2) { return []; } const values = sheet .getRange( 1, 1, lastRow, sheet.getLastColumn() ) .getValues(); const headers = values.shift(); return values.map(function(row) { const object = {}; headers.forEach(function(header, index) { object[header] = row[index]; }); return object; }); } // ===================================================== // APPEND DATA // ===================================================== function appendToSheet( sheetName, object ) { const ss = SpreadsheetApp.openById( SPREADSHEET_ID ); const sheet = ss.getSheetByName( sheetName ); if (!sheet) { throw new Error( "Sheet not found: " + sheetName ); } const headers = HEADERS[sheetName]; const row = headers.map(function(header) { return object[header] !== undefined ? object[header] : ""; }); sheet.appendRow(row); } // ===================================================== // TODAY'S DATE // ===================================================== function getToday() { return Utilities.formatDate( new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd" ); } // ===================================================== // JSON RESPONSE // ===================================================== function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
